// Zero-allocation sensor query loop
pub fn poll_thermal_sensors(hwmon: &HwmonDevice) -> Result<ThermalState> {
    let temp_mC = hwmon.read_temp_raw(Zone::CpuPackage)?;
    let target_rpm = compute_hysteresis(temp_mC, &hwmon.curve);
    hwmon.write_pwm_duty(target_rpm)?;
    Ok(ThermalState { temp_mC, target_rpm })
}
